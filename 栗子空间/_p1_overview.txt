    // —— 总览布局状态：dy = 2F 下移量（0=叠在 1F 正上方，1=落到地面），op1/op2 = 两层淡入淡出透明度 ——
    let _ovCur = null;
    let _ovAnimId = null;
    function _ovTarget() {
        const h = (isoState && isoState.hidden) || {};
        return { dy: h.f1 ? 1 : 0, op1: h.f1 ? 0 : 1, op2: h.f2 ? 0 : 1 };
    }
    function _ovEase(t) { return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2; }
    // 楼层显隐切换：垂直位移动画 + 淡入淡出；隐藏任意楼层后，剩余楼层自动落地成为完整建筑
    function _animateIsoHouse() {
        const target = _ovTarget();
        if (!_ovCur) { _ovCur = target; _renderIsoHouse(); return; }
        const from = Object.assign({}, _ovCur);
        const t0 = Date.now(), dur = 380;
        if (_ovAnimId) cancelAnimationFrame(_ovAnimId);
        const step = function () {
            const t = Math.min(1, (Date.now() - t0) / dur);
            const k = _ovEase(t);
            _ovCur = { dy: from.dy + (target.dy - from.dy) * k, op1: from.op1 + (target.op1 - from.op1) * k, op2: from.op2 + (target.op2 - from.op2) * k };
            _renderIsoHouse();
            if (t < 1) _ovAnimId = requestAnimationFrame(step);
            else { _ovCur = target; _ovAnimId = null; }
        };
        step();
    }

    // —— 花藤饰边（楼板前缘 / 屋顶边缘的一排绿叶小花） ——
    function _vineTrim(out, track, S, ax, ay, bx, by) {
        track(Math.min(ax, bx) - 6, Math.min(ay, by) - 8, Math.max(ax, bx) + 6, Math.max(ay, by) + 6);
        const n = Math.max(2, Math.round(Math.hypot(bx - ax, by - ay) / (13 * S)));
        out.push('<line x1="' + Math.round(ax) + '" y1="' + Math.round(ay - 1 * S) + '" x2="' + Math.round(bx) + '" y2="' + Math.round(by - 1 * S) + '" stroke="#8FCB9B" stroke-width="' + Math.max(2, 3.5 * S) + '" stroke-linecap="round"/>');
        for (let i = 0; i <= n; i++) {
            const t = i / n, x = ax + (bx - ax) * t, y = ay + (by - ay) * t - 1.5 * S;
            out.push('<circle cx="' + Math.round(x) + '" cy="' + Math.round(y) + '" r="' + Math.max(1.5, 3 * S) + '" fill="#A8D8BE"/>');
            if (i % 2 === 0) out.push('<circle cx="' + Math.round(x) + '" cy="' + Math.round(y - 2.5 * S) + '" r="' + Math.max(1.2, 2.1 * S) + '" fill="' + (i % 4 === 0 ? '#F9C6D9' : '#F5D48A') + '"/>');
        }
    }

    // —— 楼板 / 屋顶厚板（蛋糕胚侧面 + 顶缘高光 + 花藤）：2F 楼板与 1F 屋顶帽共用 ——
    function _pushIsoSlab(out, track, P, S, ox, oy, theme, G, night, sideH, vines, withTop, topFill) {
        const wN = night ? 0.6 : 1;
        const h = sideH * S;
        const cR = P(ox, oy, G.cols, 0), cB = P(ox, oy, G.cols, G.rows), cL = P(ox, oy, 0, G.rows);
        const face = function (a, b, fill) {
            track(a[0], a[1], a[0], a[1] + h); track(b[0], b[1], b[0], b[1] + h);
            out.push('<polygon points="' + Math.round(a[0]) + ',' + Math.round(a[1]) + ' ' + Math.round(b[0]) + ',' + Math.round(b[1]) + ' ' + Math.round(b[0]) + ',' + Math.round(b[1] + h) + ' ' + Math.round(a[0]) + ',' + Math.round(a[1] + h) + '" fill="' + fill + '" stroke="' + fill + '" stroke-width="' + Math.max(1.5, 2 * S) + '" stroke-linejoin="round"/>');
        };
        if (withTop) {
            const c0 = P(ox, oy, 0, 0);
            [c0, cR, cB, cL].forEach(function (p) { track(p[0], p[1], p[0], p[1]); });
            out.push('<polygon points="' + [c0, cR, cB, cL].map(function (p) { return Math.round(p[0]) + ',' + Math.round(p[1]); }).join(' ') + '" fill="' + topFill + '" stroke="' + topFill + '" stroke-width="' + Math.max(1.5, 2.5 * S) + '" stroke-linejoin="round"/>');
        }
        face(cL, cB, _shade(theme.wood, 1.0 * wN));  // 左前侧面（迎光）
        face(cR, cB, _shade(theme.wood, 0.86 * wN)); // 右前侧面（背光）
        out.push('<line x1="' + Math.round(cL[0]) + '" y1="' + Math.round(cL[1] + 1.5 * S) + '" x2="' + Math.round(cB[0]) + '" y2="' + Math.round(cB[1] + 1.5 * S) + '" stroke="' + _shade(theme.wood, 1.22 * wN) + '" stroke-width="' + Math.max(1.5, 2.2 * S) + '" stroke-linecap="round"/>');
        out.push('<line x1="' + Math.round(cR[0]) + '" y1="' + Math.round(cR[1] + 1.5 * S) + '" x2="' + Math.round(cB[0]) + '" y2="' + Math.round(cB[1] + 1.5 * S) + '" stroke="' + _shade(theme.wood, 1.05 * wN) + '" stroke-width="' + Math.max(1.5, 2.2 * S) + '" stroke-linecap="round"/>');
        if (vines) {
            _vineTrim(out, track, S, cL[0], cL[1], cB[0], cB[1]);
            _vineTrim(out, track, S, cB[0], cB[1], cR[0], cR[1]);
        }
    }

    function _renderIsoHouse() {
        const wrap = document.getElementById('isoHouse');
        if (!wrap) return;
        const S = 0.5;
        const G = ISO_GRID;
        const night = !!isoState.night;
        const lay = _ovCur || _ovTarget();
        // 边界自动追踪：所有图形/文字/家具都登记范围，最后按真实边界生成 1:1 画布
        let minX = 1e9, minY = 1e9, maxX = -1e9, maxY = -1e9;
        const out = [];
        const track = function (x0, y0, x1, y1) {
            minX = Math.min(minX, x0, x1); maxX = Math.max(maxX, x0, x1);
            minY = Math.min(minY, y0, y1); maxY = Math.max(maxY, y0, y1);
        };
        const P = function (ox, oy, gx, gy) { return [ox + (gx - gy) * 32 * S, oy + (gx + gy) * 16 * S]; };
        const poly = function (pts, fill, opacity, strokeW) {
            pts.forEach(function (p) { track(p[0], p[1], p[0], p[1]); });
            out.push('<polygon points="' + pts.map(function (p) { return Math.round(p[0]) + ',' + Math.round(p[1]); }).join(' ') + '" fill="' + fill + '"' + (opacity ? ' opacity="' + opacity + '"' : '') + ' stroke="' + fill + '" stroke-width="' + (strokeW || Math.max(1, 2 * S)) + '" stroke-linejoin="round"/>');
        };
        const text = function (x, y, size, str, color) {
            track(x - size * 2, y - size, x + size * 2, y + size * 0.4);
            out.push('<text x="' + Math.round(x) + '" y="' + Math.round(y) + '" font-size="' + size + '" font-weight="600" fill="' + (color || '#85532E') + '" text-anchor="middle">' + str + '</text>');
        };
        const vineTrim = function (ax, ay, bx, by) { _vineTrim(out, track, S, ax, ay, bx, by); };
        const furn = function (it, ox, oy, tk) {
            const p = P(ox, oy, it.gx + 0.5, it.gy + 0.5);
            const frag = _furnSVGHTML(it.k, S, p[0], p[1], tk, it.rot, it);
            if (frag) {
                out.push(frag);
                track(p[0] - 60 * S, p[1] - 72 * S, p[0] + 66 * S, p[1] + 22 * S);
            } else {
                text(p[0], p[1], 24, it.e || '📦');
            }
        };
        const drawSpace = function (spaceKey, ox, oy, skipSides) {
            const theme = ISO_THEMES[_spaceTheme(spaceKey)] || ISO_THEMES.cozy;
            const sp = ISO_SPACES.find(s => s.key === spaceKey);
            const PP = function (gx, gy) { return P(ox, oy, gx, gy); };
            _pushWalls(out, track, PP, S, theme, G, { stickers: _spaceTheme(spaceKey) === 'cyber', night: night });
            _pushFloor(out, track, PP, S, theme, sp, G, night);
            if (!skipSides) _pushPlatformSides(out, PP, S, theme, G, night, 14);
            // 分区隔墙
            (sp.dividers || []).forEach(function (dx) {
                const p0 = P(ox, oy, dx, 0), p1 = P(ox, oy, dx, G.rows);
                const H = 54 * S;
                poly([[p0[0], p0[1] - H], [p1[0], p1[1] - H], [p1[0], p1[1] + 2 * S], [p0[0], p0[1] + 2 * S]], theme.wallShade);
            });
            // 前墙（半透明，保留厚度感不遮挡室内）
            const H2 = 22 * S;
            const fa1 = P(ox, oy, G.cols, 0), fa2 = P(ox, oy, G.cols, G.rows);
            poly([[fa1[0], fa1[1] - H2], [fa2[0], fa2[1] - H2], [fa2[0], fa2[1]], [fa1[0], fa1[1]]], theme.frontWall);
            const fb1 = P(ox, oy, 0, G.rows), fb2 = P(ox, oy, G.cols, G.rows);
            poly([[fb1[0], fb1[1] - H2], [fb2[0], fb2[1] - H2], [fb2[0], fb2[1]], [fb1[0], fb1[1]]], theme.frontWall);
            // 名牌 + 家具
            sp.zones.forEach(function (z) {
                const p = P(ox, oy, z.c0 + (z.c1 - z.c0 + 1) / 2, z.r0 + 0.9);
                text(p[0], p[1] - 46 * S, Math.round(11 * S + 3), z.name, night ? '#F5E6C8' : '#85532E');
            });
            isoState.spaces[spaceKey].items.forEach(function (it) { furn(it, ox, oy, _spaceTheme(spaceKey)); });
        };
        // —— 建筑骨架：1F 落地；2F 水平坐标与 1F 完全重合、垂直叠在正上方；副楼右侧同一地面 ——
        const stackH = G.wallH * S; // 一层墙高（像素）
        const ox1 = 150, oy1 = 132;
        const ox2 = ox1; // 上下层严格对齐（同一栋楼，禁止错开）
        const oy2 = oy1 - stackH + lay.dy * stackH; // 隐藏 1F 时 2F 平滑下移到地面
        const oxG = ox1 + (G.cols + G.rows) * 32 * S + 20 * S; // 副楼在主楼右侧，留出小院间隙
        const oyG = oy1; // 副楼地面与主楼 1F 完全齐平
        const showF1 = lay.op1 > 0.02, showF2 = lay.op2 > 0.02;
        const showG = !!(isoState.hidden && !isoState.hidden.gaming);
        const groundTheme = ISO_THEMES[isoState.theme] || ISO_THEMES.cozy;
        const grassFill = night ? _shade(groundTheme.grass, 0.55) : groundTheme.grass;
        // —— 统一户外基底：一块连通的圆角大草坪托起主楼 + 副楼（同一个小院） ——
        const tShift = (oxG - ox1) / (64 * S);
        const m = 1.0;
        const gx0 = -m, gx1 = Math.max(G.cols, showG ? tShift + G.cols : 0) + m;
        const gy0 = (showG ? -tShift : 0) - m, gy1 = G.rows + m;
        poly([P(ox1, oy1, gx0, gy0), P(ox1, oy1, gx1, gy0), P(ox1, oy1, gx1, gy1), P(ox1, oy1, gx0, gy1)], grassFill, null, 18 * S);
        // 建筑落地阴影（柔和椭圆，右下偏移，让楼“坐”在草地上）
        if (showF1 || showF2) {
            out.push('<ellipse cx="' + Math.round(ox1 + 48) + '" cy="' + Math.round(oy1 + 82) + '" rx="152" ry="62" fill="rgba(94,58,32,0.10)"/>');
            track(ox1 - 110, oy1 + 16, ox1 + 206, oy1 + 148);
        }
        if (showG) {
            out.push('<ellipse cx="' + Math.round(oxG + 48) + '" cy="' + Math.round(oy1 + 82) + '" rx="152" ry="62" fill="rgba(94,58,32,0.10)"/>');
            track(oxG - 110, oy1 + 16, oxG + 206, oy1 + 148);
        }
        // —— 院子里的小径 / 喷泉 / 绿植（公共装饰，填充空白） ——
        const stone = function (x, y, r) {
            poly([[x, y - r * S], [x + 2 * r * S, y], [x, y + r * S], [x - 2 * r * S, y]], night ? '#B8A68C' : '#EAD9B8', null, 5 * S);
        };
        if (showG && (showF1 || showF2)) {
            [0.22, 0.5, 0.78].forEach(function (s) { stone(ox1 + 101 + 192 * s, oy1 + 28 + 96 * s, 15); }); // 主楼↔副楼 石板小径
        }
        if (showF1 || showF2) {
            [[-1.1, 4.6], [-2.1, 5.1], [-3.1, 5.6]].forEach(function (q) { const c = P(ox1, oy1, q[0], q[1]); stone(c[0], c[1], 13); }); // 门前小路
            text(ox1 + 168, oy1 + 212, 22, '⛲');   // 小喷泉
            text(ox1 - 52, oy1 + 162, 14, '🌷');
            text(ox1 + 330, oy1 + 190, 12, '🪨');
        }
        if (showG) {
            text(ox1 + 246, oy1 + 34, 16, '🌳');   // 两栋之间的绿荫
            text(ox1 + 480, oy1 + 172, 14, '🪴');
        }
        // 夜晚天空：月亮 + 星星 + 萤火虫（先画，压在建筑后面）
        if (night) {
            text(ox1 - 150, oy1 - 70, 26, '🌙');
            [[ox1 - 40, oy1 - 92, 11], [ox1 + 120, oy1 - 106, 9], [ox1 + 320, oy1 - 98, 10], [ox1 + 520, oy1 - 44, 9], [ox1 + 600, oy1 + 40, 11], [ox1 - 120, oy1 + 40, 9]].forEach(function (st) {
                text(st[0], st[1], st[2], '✦', '#F5E6C8');
            });
            [[ox1 + 40, oy1 + 180], [ox1 + 180, oy1 + 230], [ox1 + 330, oy1 + 212], [ox1 + 480, oy1 + 172]].forEach(function (f) {
                out.push('<circle cx="' + Math.round(f[0]) + '" cy="' + Math.round(f[1]) + '" r="' + Math.round(3.5 * S + 1.5) + '" fill="rgba(255, 224, 130, 0.75)"/>');
                out.push('<circle cx="' + Math.round(f[0]) + '" cy="' + Math.round(f[1]) + '" r="' + Math.round(9 * S + 3) + '" fill="rgba(255, 224, 130, 0.18)"/>');
                track(f[0] - 14, f[1] - 14, f[0] + 14, f[1] + 14);
            });
        }
        // —— 主楼 1F：接地；隐藏 2F 时顶部自动盖上屋顶厚板收口，仍是完整建筑 ——
        if (showF1) {
            out.push('<g opacity="' + lay.op1.toFixed(3) + '">');
            drawSpace('f1', ox1, oy1);
            if (lay.op2 < 0.97) {
                out.push('<g opacity="' + Math.min(1, (1 - lay.op2) * 1.6).toFixed(3) + '">');
                _pushIsoSlab(out, track, P, S, ox1, oy1 - stackH, ISO_THEMES[_spaceTheme('f1')] || groundTheme, G, night, 10, true, true, night ? _shade(groundTheme.roof, 0.6) : groundTheme.roof);
                out.push('</g>');
            }
            out.push('</g>');
        }
        // —— 主楼 2F：严格垂直叠在 1F 正上方（中间夹实体楼板）；隐藏 1F 时整体下移落地 ——
        if (showF2) {
            out.push('<g opacity="' + lay.op2.toFixed(3) + '">');
            _pushIsoSlab(out, track, P, S, ox2, oy2, ISO_THEMES[_spaceTheme('f2')] || groundTheme, G, night, 16, true, false);
            drawSpace('f2', ox2, oy2, true);
            // 楼顶绿化：沿 2F 背墙顶的一排花藤
            const rA = P(ox2, oy2, 0, 0), rB = P(ox2, oy2, G.cols, 0), rC = P(ox2, oy2, 0, G.rows);
            const roofLift = stackH + 8 * S;
            vineTrim(rA[0], rA[1] - roofLift, rB[0], rB[1] - roofLift);
            vineTrim(rA[0], rA[1] - roofLift, rC[0], rC[1] - roofLift);
            out.push('</g>');
        }
        // —— 副楼（电竞房）：独立单层，落地在主楼右侧，共享同一片草地 ——
        if (showG) {
            drawSpace('gaming', oxG, oyG);
            const ga = P(oxG, oyG, 0, 0), gb = P(oxG, oyG, G.cols, 0), gc = P(oxG, oyG, 0, G.rows);
            const gh = stackH + 7 * S;
            vineTrim(ga[0], ga[1] - gh, gb[0], gb[1] - gh);
            vineTrim(ga[0], ga[1] - gh, gc[0], gc[1] - gh);
        }
        // 总览里的小居民：白天在院里溜达，夜晚回二楼睡觉（💤）
        {
            const homeUids = Object.keys(appData.users || {}).slice(0, 2);
            if (!night) {
                homeUids.forEach(function (uid, i) {
                    const c = P(ox1, oy1, 2.2 + i * 2.4, G.rows + 1.3 + (i % 2) * 0.5);
                    text(c[0], c[1], 17, getAvatarEmoji(uid) || '🌰');
                });
            } else if (showF2) {
                const zc = P(ox2, oy2, 3.4, 3.2);
                text(zc[0], zc[1] - 34 * S, 13, '💤');
            }
        }
        // 生成 1:1 像素 SVG（热点层与之共用坐标系，绝不错位）；容器窄时整体等比缩放
        const w = Math.ceil(maxX - minX), h = Math.ceil(maxY - minY);
        const svg = '<svg xmlns="http://www.w3.org/2000/svg" width="' + w + '" height="' + h + '" viewBox="' + Math.floor(minX) + ' ' + Math.floor(minY) + ' ' + w + ' ' + h + '" style="display:block">' + out.join('') + '</svg>';
        _lastOverviewSvg = svg;
        const spots = [];
        if (showF2 && lay.op2 > 0.5) spots.push(['f2', ox2, oy2]);
        if (showF1 && lay.op1 > 0.5) spots.push(['f1', ox1, oy1]);
        if (showG) spots.push(['gaming', oxG, oyG]);
        const hotspots = spots.map(function (pair) {
            const tl = P(pair[1], pair[2], 0, 0), r = P(pair[1], pair[2], G.cols, 0), b = P(pair[1], pair[2], G.cols, G.rows), l = P(pair[1], pair[2], 0, G.rows);
            const lx = Math.min(tl[0], l[0]) - 4, rx = Math.max(r[0], b[0]) + 4;
            const ty = tl[1] - ISO_GRID.wallH * S - 10, by = b[1] + 26 * S;
            return '<div class="house-hotspot" data-space="' + pair[0] + '" style="left:' + Math.round(lx - minX) + 'px;top:' + Math.round(ty - minY) + 'px;width:' + Math.round(rx - lx) + 'px;height:' + Math.round(by - ty) + 'px" title="点击布置"></div>';
        }).join('');
        const cw = (wrap.clientWidth || 380) - 4;
        const k = Math.min(1, cw / w);
        // 画布全天保持明亮奶白（App 深色模式也不压暗小屋，像暗房里亮着的模型灯）；夜晚模式除外
        wrap.style.background = night
            ? 'linear-gradient(180deg, #2A2E55 0%, #3A4070 55%, #4A5188 100%)'
            : 'linear-gradient(180deg, #FDF6E7 0%, #FAEEDA 100%)';
        wrap.innerHTML = '<div style="position:relative;width:' + Math.round(w * k) + 'px;height:' + Math.round(h * k) + 'px;margin:0 auto;">' +
            '<div style="position:absolute;left:0;top:0;width:' + w + 'px;height:' + h + 'px;transform:scale(' + k + ');transform-origin:top left;">' + svg + hotspots + '</div>' +
            '</div>';
        wrap.querySelectorAll('.house-hotspot').forEach(function (el) {
            el.addEventListener('click', function () { selectIsoSpace(el.dataset.space); });
        });
    }
